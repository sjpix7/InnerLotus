chakra.root.name = Racine
chakra.root.sanskrit = Muladhara
chakra.root.core = JE SUIS
chakra.root.location = Base de la colonne vertébrale
chakra.root.element = Terre
chakra.root.bija = LAM
chakra.root.theme = Sécurité, ancrage, stabilité, appartenance
chakra.root.balanced = Se sentir en sécurité, ancré, stable et présent dans son corps.
chakra.root.imbalanced = Peur, anxiété, soucis d'argent, agitation, sentiment de déconnexion.
chakra.root.practice = Marchez pieds nus sur la terre, mangez des légumes racines, prenez la posture de la montagne et chantez LAM.
chakra.root.seeds:
- Je suis en sécurité, {centré|centrée} et {ancré|ancrée}.
- Je suis {enraciné|enracinée} comme un arbre ancien.
- La terre me soutient à chaque pas.
- J'ai confiance que tous mes besoins sont comblés.
- J'ai ma place ici, dans ce corps et dans cet instant.
- Je me tiens stable et {fort|forte}.

chakra.sacral.bija = VAM
chakra.sacral.theme = Émotions, créativité, plaisir, fluidité

- Ma créativité coule avec aisance.
- J'honore mes désirs et mon corps.
- J'accueille le changement avec grâce.
