chakra.root.name = Radice
chakra.root.sanskrit = Muladhara
chakra.root.core = IO SONO
chakra.root.location = Base della colonna vertebrale
chakra.root.element = Terra
chakra.root.bija = LAM
chakra.root.theme = Sicurezza, radicamento, stabilità, appartenenza
chakra.root.balanced = Sentirsi al sicuro, radicati, stabili e presenti nel proprio corpo.
chakra.root.imbalanced = Paura, ansia, preoccupazioni economiche, irrequietezza, senso di distacco.
chakra.root.practice = Cammina a piedi nudi sulla terra, mangia ortaggi a radice, assumi la posizione della montagna e canta LAM.
chakra.root.seeds:
- Sono al sicuro, {centrato|centrata} e {radicato|radicata}.
- Ho radici profonde come un albero antico.
- La terra mi sostiene a ogni passo.
- Mi fido che tutti i miei bisogni sono soddisfatti.
- Appartengo a questo luogo, a questo corpo e a questo momento.
- Sto {saldo|salda} e forte.
